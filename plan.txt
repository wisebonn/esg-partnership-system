function dupKey(title: string, period: string | null): string {
  const t = title.trim().toLowerCase();
  const p = period ?? "N/A";
  return t + "|" + p;
}

function getTimelineColor(task: Task) {
  if (!task.due_date) return { text: "No deadline", color: "text-slate-500", bg: "bg-slate-800", priority: 5 };
  if (task.status === "Verified") return { text: "Completed", color: "text-emerald-500", bg: "bg-emerald-500/20", priority: 6 };
  if (task.status === "Pending Evidence") return { text: "Awaiting Approval", color: "text-blue-400", bg: "bg-blue-500/20", priority: 5 };
  const start = new Date(task.start_date || task.created_at).getTime();
  const due = new Date(task.due_date).getTime();
  const now = new Date().getTime();
  const totalTime = due - start;
  const timeLeft = due - now;
  if (timeLeft < 0) return { text: "Overdue", color: "text-red-500 font-bold", bg: "bg-red-500/20", priority: 1 };
  const ratio = timeLeft / totalTime;
  if (ratio <= 0.25) return { text: "Critical", color: "text-red-500", bg: "bg-red-500/20", priority: 2 };
  if (ratio <= 0.50) return { text: "Approaching Halfway", color: "text-orange-500", bg: "bg-orange-500/20", priority: 3 };
  return { text: "On Track", color: "text-emerald-500", bg: "bg-emerald-500/20", priority: 4 };
}

function formatCountdown(dueDateStr: string | null, status: string) {
  if (!dueDateStr) return "";
  if (status === "Verified" || status === "Pending Evidence") return "";
  const due = new Date(dueDateStr).getTime();
  const now = new Date().getTime();
  const diff = due - now;
  if (diff < 0) return "Time expired";
  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
  const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
  const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
  if (days > 0) return days + "d " + hours + "h left";
  if (hours > 0) return hours + "h " + minutes + "m left";
  return minutes + "m left";
}

function isTaskOnDate(task: Task, targetDate: Date) {
  if (!task.due_date) return false;
  const due = new Date(task.due_date);
  if (due.toDateString() === targetDate.toDateString()) return true;
  if (!task.recurrence_rule) return false;
  if (targetDate < new Date(due.getFullYear(), due.getMonth(), due.getDate())) return false;

  if (task.recurrence_rule === "weekly") {
    const diffTime = targetDate.getTime() - due.getTime();
    const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
    if (diffDays > 0 && diffDays % 7 === 0) return true;
  }
  if (task.recurrence_rule === "monthly") {
    if (targetDate.getDate() === due.getDate() && targetDate > due) return true;
  }
  if (task.recurrence_rule === "quarterly") {
    if (targetDate.getDate() === due.getDate() && (targetDate.getMonth() - due.getMonth()) % 3 === 0 && targetDate > due) return true;
  }
  return false;
}

              <div className="flex gap-3 pt-2"><button type="button" onClick={() => setIsModalOpen(false)} className="flex-1 py-2.5 bg-slate-800 text-white rounded-lg font-medium">Cancel</button><button type="submit" disabled={saving} className="flex-1 py-2.5 bg-blue-600 text-white rounded-lg font-medium disabled:opacity-50">{saving ? <Loader2 className="animate-spin" size={18} /> : "Save"}</button></div>
            </form>
          </div>
        </div>
      )}

      {editTask && (
        <div className="fixed inset-0 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4 z-50">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-lg shadow-2xl p-6 space-y-4">
            <div className="flex justify-between items-start">
              <h2 className="text-lg font-semibold text-white pr-4">{editTask.title}</h2>
              <button onClick={() => setEditTask(null)} className="text-slate-400 hover:text-white mt-1"><X size={20} /></button>
            </div>
            
            {editTask.description && <p className="text-sm text-slate-400">{editTask.description}</p>}
            
            <div className="grid grid-cols-2 gap-3 pt-2">
              <div>
                <label className="block text-xs font-medium text-slate-400 mb-1">Status</label>
                <select 
                  value={editTask.status} 
                  onChange={(e) => {
                    updateStatus(editTask.id, e.target.value);
                    setEditTask({...editTask, status: e.target.value});
                  }} 
                  className={formCls}
                >
                  {["To Do", "In Progress", "Blocked", "Pending Evidence", "Verified"].map(s => <option key={s} value={s}>{s}</option>)}
                </select>
              </div>
              <div>
                <label className="block text-xs font-medium text-slate-400 mb-1">Due Date</label>
                <input type="datetime-local" value={editForm.due_date} onChange={(e) => setEditForm({ ...editForm, due_date: e.target.value })} className={formCls} />
              </div>
            </div>

            {canManage && (
              <div className="pt-2">
                <label className="block text-xs font-medium text-slate-400 mb-1">Assignee (Start on behalf of)</label>
                <select value={editForm.assigned_to} onChange={(e) => setEditForm({ ...editForm, assigned_to: e.target.value })} className={formCls}>
                  <option value="">Unassigned</option>
                  {profiles.map(p => <option key={p.id} value={p.id}>{p.full_name}</option>)}
                </select>
              </div>
            )}

            <div className="flex flex-col gap-2 pt-4 border-t border-slate-800">
              {editTask.status !== "Verified" && (
                <button 
                  onClick={() => {
                    setEvidenceTask(editTask);
                    setEditTask(null);
                  }} 
                  className="w-full py-2.5 bg-emerald-600 hover:bg-emerald-500 text-white rounded-lg font-medium flex items-center justify-center gap-2"
                >
                  <CheckCircle2 size={18} /> Mark Complete / Upload Evidence
                </button>
              )}
              {editTask.status === "To Do" && (
                <button 
                  onClick={() => {
                    updateStatus(editTask.id, "In Progress");
                    setEditTask(null);
                  }} 
                  className="w-full py-2.5 bg-blue-600 hover:bg-blue-500 text-white rounded-lg font-medium flex items-center justify-center gap-2"
                >
                  <Clock size={18} /> Start Task
                </button>
              )}
              <button 
                type="button" 
                onClick={handleEditSubmit} 
                disabled={saving} 
                className="w-full py-2.5 bg-slate-800 hover:bg-slate-700 text-white rounded-lg font-medium disabled:opacity-50"
              >
                {saving ? <Loader2 className="animate-spin" size={18} /> : "Save Changes"}
              </button>
            </div>
          </div>
        </div>
      )}

      {evidenceTask && (
        <div className="fixed inset-0 bg-black/70 flex items-center justify-center p-4 z-50">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-lg shadow-2xl p-6">
            <h2 className="text-lg font-semibold text-white mb-2">Mark Task Complete</h2>
            <p className="text-sm text-slate-400 mb-4">
              {isManager ? "As a Manager, your submission will be automatically verified." : "Upload evidence for your manager to verify."}
            </p>
            <form onSubmit={markComplete} className="space-y-4">
              <div><label className="block text-sm font-medium text-slate-300 mb-1">Evidence Link (Optional)</label><input type="url" value={evidenceLink} onChange={(e) => setEvidenceLink(e.target.value)} className={formCls} /></div>
              <div><label className="block text-sm font-medium text-slate-300 mb-1">Note (Optional)</label><textarea rows={3} value={evidenceComment} onChange={(e) => setEvidenceComment(e.target.value)} className={formCls} /></div>
              <div className="flex gap-3 pt-2">
                <button type="button" onClick={() => setEvidenceTask(null)} className="flex-1 py-2.5 bg-slate-800 text-white rounded-lg font-medium">Cancel</button>
                <button type="submit" disabled={submittingEvidence} className="flex-1 py-2.5 bg-emerald-600 text-white rounded-lg font-medium disabled:opacity-50">
                  {submittingEvidence ? <Loader2 className="animate-spin" size={18} /> : (isManager ? "Verify & Complete" : "Submit for Approval")}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {activeTaskForComments && me && (
        <TaskCommentsModal taskId={activeTaskForComments.id} taskTitle={activeTaskForComments.title} currentUserId={me.id} onClose={() => { setActiveTaskForComments(null); fetchAll(); }} />
      )}

      {showTimelineModal && needsTimeline.length > 0 && (
        <div className="fixed inset-0 bg-black/70 flex items-center justify-center p-4 z-50">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-2xl shadow-2xl p-6">
            <h2 className="text-lg font-semibold text-white mb-2">Timeline Needed</h2>
            <p className="text-sm text-slate-400 mb-4">The following tasks have no period or date. Please assign them so they can be scheduled.</p>
            <div className="max-h-60 overflow-y-auto space-y-3 mb-6">
              {needsTimeline.map(n => (
                <div key={n.id} className="flex items-center gap-4 bg-slate-950 p-3 rounded-lg border border-slate-800">
                  <span className="flex-1 text-sm text-slate-200 truncate">{n.title}</span>
                  <select value={timelineDraft[n.id]?.period || "Q1"} onChange={(e) => setTimelineDraft(prev => ({ ...prev, [n.id]: { ...prev[n.id], period: e.target.value as TaskPeriod } }))} className="bg-slate-800 text-sm text-white p-1 rounded">
                    {PERIODS.filter(p => p !== "N/A").map(p => <option key={p} value={p}>{p}</option>)}
                  </select>
                  <input type="date" value={timelineDraft[n.id]?.due_date || ""} onChange={(e) => setTimelineDraft(prev => ({ ...prev, [n.id]: { ...prev[n.id], due_date: e.target.value } }))} className="bg-slate-800 text-sm text-white p-1 rounded" />
                </div>
              ))}
            </div>
            <div className="flex justify-end gap-2">
              <button onClick={() => setShowTimelineModal(false)} className="px-4 py-2 text-sm text-slate-300 hover:text-white">Cancel</button>
              <button onClick={applyTimelineDrafts} className="px-4 py-2 text-sm bg-blue-600 text-white rounded-lg hover:bg-blue-500">Save & Schedule</button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
